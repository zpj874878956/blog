# 安装krew

二进制安装

```bash
wget https://github.com/itaysk/kubectl-neat/releases/download/v2.0.4/kubectl-neat_linux_amd64.tar.gz

tar -zxvf kubectl-neat_linux_amd64.tar.gz

chmod +x kubectl-neat

mv kubectl-neat /usr/local/bin/
```

临时生效：

```bash
export PATH=/usr/local/bin:$PATH
```

永久生效：

```bash
vi /root/.bashrc
export PATH=/usr/local/bin:$PATH
source /root/.bashrc
```

# 批量精简脚本

```bash
#cat ingress-neat.sh
#!/bin/bash

# ==========================================
# Kubernetes Ingress 导出并精简脚本
# 功能：
#   1. 获取指定命名空间下所有 Ingress
#   2. 使用 kubectl neat 精简 YAML
#   3. 每个 Ingress 单独保存为 YAML 文件
#
# 使用：
#   bash export_ingress.sh /data/k8s-backup/ingress
# ==========================================

set -e

# 命名空间
NAMESPACE="cc"

# 输出目录
OUTPUT_DIR="${1:-./ingress-backup}"

# 日志函数
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*"
}

# 检查 kubectl
if ! command -v kubectl >/dev/null 2>&1; then
    echo "ERROR: 未找到 kubectl 命令"
    exit 1
fi

# 检查 kubectl neat
if ! kubectl neat --help >/dev/null 2>&1; then
    echo "ERROR: kubectl neat 不可用"
    echo "请确认 kubectl-neat 已安装并且所在目录已经加入 PATH"
    exit 1
fi

# 检查 Kubernetes 连接
if ! kubectl get namespace "${NAMESPACE}" >/dev/null 2>&1; then
    echo "ERROR: 无法访问 Kubernetes，或者命名空间 ${NAMESPACE} 不存在"
    exit 1
fi

# 创建输出目录
mkdir -p "${OUTPUT_DIR}"

log "开始导出命名空间 ${NAMESPACE} 下的 Ingress"
log "输出目录：${OUTPUT_DIR}"

# 获取所有 Ingress 名称
INGRESS_LIST=$(kubectl get ingress -n "${NAMESPACE}" \
    -o jsonpath='{range .items[*]}{.metadata.name}{"\n"}{end}')

# 判断是否存在 Ingress
if [ -z "${INGRESS_LIST}" ]; then
    log "命名空间 ${NAMESPACE} 下没有 Ingress"
    exit 0
fi

SUCCESS_COUNT=0
FAILED_COUNT=0

while IFS= read -r INGRESS_NAME
do
    [ -z "${INGRESS_NAME}" ] && continue

    OUTPUT_FILE="${OUTPUT_DIR}/${INGRESS_NAME}.yaml"

    log "正在导出：${INGRESS_NAME}"

    if kubectl get ingress "${INGRESS_NAME}" \
        -n "${NAMESPACE}" \
        -o yaml \
        | kubectl neat \
        > "${OUTPUT_FILE}"; then

        log "导出成功：${OUTPUT_FILE}"
        SUCCESS_COUNT=$((SUCCESS_COUNT + 1))

    else
        log "导出失败：${INGRESS_NAME}"

        # 删除失败产生的空文件
        rm -f "${OUTPUT_FILE}"

        FAILED_COUNT=$((FAILED_COUNT + 1))
    fi

done <<< "${INGRESS_LIST}"

echo
echo "=========================================="
echo "Ingress 导出完成"
echo "命名空间：${NAMESPACE}"
echo "输出目录：${OUTPUT_DIR}"
echo "成功数量：${SUCCESS_COUNT}"
echo "失败数量：${FAILED_COUNT}"
echo "=========================================="

exit 0
```

# 删除yaml文件多余的参数，这里是annotations

```bash
# cat remove_annotations.sh
#!/bin/bash

# ==========================================
# 批量删除 YAML 中 metadata.annotations
#
# 使用：
#   ./remove_annotations.sh /data/k8s-backup/ingress
#
# 示例：
#   ./remove_annotations.sh ./ingress-backup
# ==========================================

set -e

YAML_DIR="${1:-.}"

# 检查目录
if [ ! -d "${YAML_DIR}" ]; then
    echo "ERROR: 目录不存在: ${YAML_DIR}"
    exit 1
fi

# 检查 yq
if ! command -v yq >/dev/null 2>&1; then
    echo "ERROR: 未找到 yq 命令"
    echo "请先安装 yq v4"
    exit 1
fi

echo "=========================================="
echo "YAML目录: ${YAML_DIR}"
echo "操作: 删除 metadata.annotations"
echo "=========================================="

SUCCESS=0
FAILED=0

# 查找 yaml / yml 文件
while IFS= read -r -d '' FILE
do
    echo "处理: ${FILE}"

    # 原地修改
    if yq -i 'del(.metadata.annotations)' "${FILE}"; then
        echo "  [OK]"
        SUCCESS=$((SUCCESS + 1))
    else
        echo "  [FAILED]"
        FAILED=$((FAILED + 1))
    fi

done < <(find "${YAML_DIR}" -type f \( -name "*.yaml" -o -name "*.yml" \) -print0)

echo
echo "=========================================="
echo "处理完成"
echo "成功: ${SUCCESS}"
echo "失败: ${FAILED}"
echo "=========================================="
```

#!/bin/bash

YAML_DIR="${1:-.}"
SECRET_NAME="7moor-com-ssl"

if ! command -v yq >/dev/null 2>&1; then
    echo "ERROR: 未找到 yq 命令"
    exit 1
fi

if [ ! -d "$YAML_DIR" ]; then
    echo "ERROR: 目录不存在: $YAML_DIR"
    exit 1
fi

for file in "$YAML_DIR"/*.yaml; do

    [ -f "$file" ] || continue

    echo "处理: $file"

    HOST=$(yq '.spec.rules[0].host' "$file")

    if [ -z "$HOST" ] || [ "$HOST" = "null" ]; then
        echo "  跳过：没有找到 spec.rules[0].host"
        continue
    fi

    echo "  域名: $HOST"

    yq -i "
      .spec.tls = [
        {
          \"hosts\": [\"$HOST\"],
          \"secretName\": \"$SECRET_NAME\"
        }
      ]
    " "$file"

    echo "  TLS 添加完成"
done

echo "全部处理完成"
